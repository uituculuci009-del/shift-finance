        function formatDateKey(d) {
            let m = '' + (d.getMonth() + 1), day = '' + d.getDate(), y = d.getFullYear();
            if (m.length < 2) m = '0' + m;
            if (day.length < 2) day = '0' + day;
            return [y, m, day].join('-');

        function getShiftDataForMonth() {
            const pattern = document.getElementById('shift-pattern').value;
            const startDateInput = document.getElementById('shift-start-date').value;
            if (!startDateInput) return [];
            const startDate = new Date(startDateInput);
            startDate.setHours(0,0,0,0);
            const totalDays = new Date(currentYear, currentMonth + 1, 0).getDate();
            let shifts = [];
            for (let d = 1; d <= totalDays; d++) {
                let currDate = new Date(currentYear, currentMonth, d);
                currDate.setHours(0,0,0,0);
                let dateKey = formatDateKey(currDate);
                let isWork = false, isNight = false, shiftLabel = "Liber";
                let diffDays = Math.round((currDate - startDate) / (1000 * 60 * 60 * 24));
                if (pattern === '4-4') {
                    let pos = ((diffDays % 8) + 8) % 8;
                    isWork = pos < 4;
                    shiftLabel = isWork ? "Lucru" : "Liber";
                } else if (pattern === '3-4') {
                    let pos = ((diffDays % 7) + 7) % 7;
                    isWork = pos < 3;
                    shiftLabel = isWork ? "Lucru" : "Liber";
                } else if (pattern === '5-2') {
                    let dayOfWeek = currDate.getDay();
                    isWork = (dayOfWeek !== 0 && dayOfWeek !== 6);
                    shiftLabel = isWork ? "Lucru" : "Liber";
                } else if (pattern === '12-24') {
                    let pos = ((diffDays % 4) + 4) % 4;
                    if (pos === 0) { isWork = true; shiftLabel = "Zi 12h"; }
                    else if (pos === 1) { isNight = true; isWork = true; shiftLabel = "Noapte"; }
                }
                if (dayOverrides[dateKey]) {
                    let ov = dayOverrides[dateKey];
                    if (ov === 'work') { isWork = true; shiftLabel = "Zi"; }
                    if (ov === 'night') { isWork = true; isNight = true; shiftLabel = "Noapte"; }
                    if (ov === 'off') { isWork = false; shiftLabel = "Liber"; }
                }
                if (isWork) { shifts.push({ date: currDate, label: shiftLabel }); }
            }
            return shifts;
        }

        function exportICS() {
            const shifts = getShiftDataForMonth();
            if (shifts.length === 0) {
                alert("Nu există ture de lucru programate în această lună pentru export.");
                return;
            }
            let icsContent = "BEGIN:VCALENDAR\nVERSION:2.0\nPROID:-//ShiftManager//RO\nCALSCALE:GREGORIAN\n";
            shifts.forEach(s => {
                let startYMD = formatDateKey(s.date).replace(/-/g, '');
                let nextDay = new Date(s.date);
                nextDay.setDate(nextDay.getDate() + 1);
                let endYMD = formatDateKey(nextDay).replace(/-/g, '');
                icsContent += "BEGIN:VEVENT\n";
                icsContent += "DTSTART;VALUE=DATE:" + startYMD + "\n";
                icsContent += "DTEND;VALUE=DATE:" + endYMD + "\n";
                icsContent += "SUMMARY:Tură - " + s.label + "\n";
                icsContent += "DESCRIPTION:Programat prin intermediul aplicației Shift Manager\n";
                icsContent += "END:VEVENT\n";
            });
            icsContent += "END:VCALENDAR";
            let blob = new Blob([icsContent], { type: 'text/calendar;charset=utf-8' });
            let a = document.createElement('a');
            a.href = URL.createObjectURL(blob);
            a.download = 'Ture_Lucru_' + currentYear + '_' + (currentMonth + 1) + '.ics';
            a.click();
        }
